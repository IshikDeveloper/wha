$oemKey = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\OEMInformation"
New-Item -Path $oemKey -Force | Out-Null
Set-ItemProperty -Path $oemKey -Name "Manufacturer" -Value "YourBrand"
Set-ItemProperty -Path $oemKey -Name "Model" -Value "Custom Build"
Set-ItemProperty -Path $oemKey -Name "SupportPhone" -Value "+1-555-0100"
Set-ItemProperty -Path $oemKey -Name "SupportURL" -Value "https://yourbrand.example.com/support"
Set-ItemProperty -Path $oemKey -Name "SupportHours" -Value "9am-5pm Mon-Fri"
Set-ItemProperty -Path $oemKey -Name "Logo" -Value "C:\Windows\System32\oobe\info\OEMLOGO.bmp"
